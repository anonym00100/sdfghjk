from flask import Flask, render_template_string
import sqlite3
import os

app = Flask(__name__)

def db():
    return sqlite3.connect("bot.db")

@app.route("/")
def home():
    conn = db()
    cur = conn.cursor()

    cur.execute("SELECT name, balance, orders FROM users ORDER BY balance DESC")
    users = cur.fetchall()

    html = """
    <h1>📊 Affiliate Dashboard</h1>
    <table border="1" cellpadding="10">
        <tr>
            <th>Name</th>
            <th>Balance</th>
            <th>Orders</th>
        </tr>
        {% for u in users %}
        <tr>
            <td>{{u[0]}}</td>
            <td>{{u[1]}}$</td>
            <td>{{u[2]}}</td>
        </tr>
        {% endfor %}
    </table>
    """

    return render_template_string(html, users=users)


# ======================
# RENDER FIXED PORT
# ======================
if __name__ == "__main__":
    port = int(os.environ.get("PORT", 10000))
    app.run(host="0.0.0.0", port=port)
